#### 1.1 `_handle_compress_session_stream(self, request)` handler

```python
# plugin.py 末尾, 跟 P15.2 同位置, 加 P16

import asyncio
import json
import time
from aiohttp import web

# P16 (6/17 鸿波) — Companion 主动 trigger hermes compress + SSE 进度
#
# 真因背景: P3.5.17.b 修了 hermes 自带 preflight 压缩, 但鸿波想要的是
# Companion 检测到 80%+ 主动调而不是被动等下次发消息时压. + 进度弹窗.
#
# 实现: 抄 Slack _handle_compress_command (gateway/run.py:13117) 套 SSE.
# 不重写压缩 — compress_context() + replace_messages() 都现成.

async def _handle_compress_session_stream(self, request):
    """POST /api/sessions/{session_id}/compress/stream

    Body (JSON, optional): {"focus_topic": "...", "force": true|false}
    Response: SSE 事件流
      - event: compress.started      data: {messages_count, approx_tokens}
      - event: compress.progress     data: {kind: "compressing|status", text}
      - event: compress.warn         data: {text}
      - event: compress.completed    data: {headline, token_line, note, before_count, after_count}
      - event: compress.failed       data: {error}
    """
    session_id = request.match_info["session_id"]

    # 走跟 _handle_session_chat_stream 同款 auth
    # (X-Hermes-API-Key / Authorization: Bearer 都接受 — abstract 在 self._check_auth)
    if not self._check_auth(request):
        return web.json_response({"error": "unauthorized"}, status=401)

    # SSE setup
    resp = web.StreamResponse(
        status=200,
        headers={
            "Content-Type": "text/event-stream",
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )
    await resp.prepare(request)

    async def send_event(event: str, data: dict):
        line = f"event: {event}\ndata: {json.dumps(data, ensure_ascii=False)}\n\n"
        await resp.write(line.encode("utf-8"))

    try:
        # 1. 从 SessionDB load session + messages
        db = self._ensure_session_db()
        session_row = db.get_session(session_id)
        if session_row is None:
            await send_event("compress.failed", {"error": f"session not found: {session_id}"})
            await resp.write_eof()
            return resp

        messages = db.get_messages(session_id)
        if len(messages) < 4:
            await send_event("compress.failed", {
                "error": f"too few messages to compress ({len(messages)} < 4)",
            })
            await resp.write_eof()
            return resp

        # 2. 估 token + emit started
        from agent.model_metadata import estimate_request_tokens_rough
        model_name = session_row.get("model") or "catfish-private-main"
        system_prompt = session_row.get("system_prompt") or ""
        approx_tokens = estimate_request_tokens_rough(
            messages, system_prompt=system_prompt, tools=None,
        )
        before_count = len(messages)
        await send_event("compress.started", {
            "messages_count": before_count,
            "approx_tokens": approx_tokens,
            "model": model_name,
        })

        # 3. body 解析 (可选 focus_topic / force)
        try:
            body = await request.json() if request.can_read_body else {}
        except Exception:
            body = {}
        focus_topic = str(body.get("focus_topic", "") or "").strip() or None
        force = bool(body.get("force", False))

        # 4. 临时 AIAgent (跟 Slack /compress 同款 tmp_agent pattern)
        from agent.run_agent import AIAgent
        from agent.conversation_compression import compress_context
        from agent.manual_compression_feedback import summarize_manual_compression

        # status_callback 桥: agent._emit_status / _emit_warning → SSE
        status_queue = asyncio.Queue()
        loop = asyncio.get_event_loop()

        def status_callback(kind: str, text: str = ""):
            # 跨线程: AIAgent._compress_context 在 executor 跑, callback 也在 executor 线程
            asyncio.run_coroutine_threadsafe(
                status_queue.put((kind, text)), loop,
            )

        tmp_agent = AIAgent(
            session_id=session_id,
            model_name=model_name,
            system_prompt=system_prompt,
            status_callback=status_callback,
            # SessionDB 复用, 让 compress_context 写回同一 db
            session_db=db,
        )

        # 5. 跑 compression in executor + drain status_queue 推 SSE
        compress_future = loop.run_in_executor(
            None,
            lambda: compress_context(
                tmp_agent, messages, system_prompt,
                approx_tokens=approx_tokens,
                focus_topic=focus_topic,
                force=force,
            ),
        )

        async def drain_status():
            while True:
                try:
                    kind, text = await asyncio.wait_for(status_queue.get(), timeout=0.5)
                except asyncio.TimeoutError:
                    if compress_future.done():
                        return
                    continue
                if kind == "warn":
                    await send_event("compress.warn", {"text": text})
                else:
                    await send_event("compress.progress", {"kind": kind, "text": text})

        # 并发 drain + 等结果
        drain_task = asyncio.create_task(drain_status())
        try:
            compressed_messages, _summary_text = await compress_future
        finally:
            drain_task.cancel()
            try:
                await drain_task
            except asyncio.CancelledError:
                pass

        # 6. 写回 SessionDB (跟 /fork pattern 同)
        db.replace_messages(session_id, compressed_messages)

        # 7. summary + emit completed
        after_count = len(compressed_messages)
        summary = summarize_manual_compression(
            before_messages=messages,
            after_messages=compressed_messages,
            before_tokens=approx_tokens,
            # after_tokens 估算
            after_tokens=estimate_request_tokens_rough(
                compressed_messages, system_prompt=system_prompt, tools=None,
            ),
            focus_topic=focus_topic,
        )
        await send_event("compress.completed", {
            "before_count": before_count,
            "after_count": after_count,
            **summary,  # {headline, token_line, note, noop}
        })

    except Exception as e:
        logger.exception("P16 _handle_compress_session_stream 异常")
        try:
            await send_event("compress.failed", {"error": str(e)})
        except Exception:
            pass

    await resp.write_eof()
    return resp


def _patch_p16_compress_endpoint() -> None:
    """P16: 加 POST /api/sessions/{session_id}/compress/stream SSE endpoint."""
    from gateway.platforms.api_server import APIServerAdapter

    APIServerAdapter._handle_compress_session_stream = _handle_compress_session_stream
    logger.info("P16 APIServerAdapter._handle_compress_session_stream 已挂 ✓")

    # 注册 route 在 connect() wrap 时
    # (跟 P7 _proxy_404_middleware 同位置, plugin.py:1235-1246)
    _orig_connect = APIServerAdapter.connect

    async def patched_connect(self):
        result = await _orig_connect(self)
        # 挂 route
        if self._app is not None:
            self._app.router.add_post(
                "/api/sessions/{session_id}/compress/stream",
                lambda req: self._handle_compress_session_stream(req),
            )
            logger.info("P16 route /api/sessions/{id}/compress/stream 已注册 ✓")
        return result

    # 注意: 之前 P7 已经 wrap connect, 这里要叠加 wrap, 不是替换
    APIServerAdapter.connect = patched_connect
```

#### 2.1 `src/lib/hermesRpc.ts` (新, ~100 行)

```typescript
/** P3.5.18 — hermes 8642 SSE client for /api/sessions/{id}/compress/stream */

export type CompressEvent =
  | { event: "compress.started"; data: { messages_count: number; approx_tokens: number; model: string } }
  | { event: "compress.progress"; data: { kind: string; text: string } }
  | { event: "compress.warn"; data: { text: string } }
  | { event: "compress.completed"; data: { before_count: number; after_count: number; headline: string; token_line: string; note: string; noop: boolean } }
  | { event: "compress.failed"; data: { error: string } };

export interface CompressRequest {
  sessionId: string;
  focusTopic?: string;
  force?: boolean;
  hermesUrl: string;  // backendUrl
  hermesAuth: string; // Bearer ...
  userEmail: string;
}

export async function* streamCompress(
  req: CompressRequest,
  signal?: AbortSignal,
): AsyncGenerator<CompressEvent, void, void> {
  const url = `${req.hermesUrl}/api/sessions/${encodeURIComponent(req.sessionId)}/compress/stream`;
  const body = JSON.stringify({ focus_topic: req.focusTopic, force: req.force });
  const resp = await fetch(url, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Accept": "text/event-stream",
      "Authorization": req.hermesAuth,
      "X-Catfish-User": req.userEmail,
    },
    body,
    signal,
  });
  if (!resp.ok || !resp.body) {
    throw new Error(`compress stream failed: HTTP ${resp.status}`);
  }
  const reader = resp.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    // SSE 帧: 双换行分割
    let idx;
    while ((idx = buffer.indexOf("\n\n")) !== -1) {
      const frame = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 2);
      const event = frame.match(/^event:\s*(.+)$/m)?.[1]?.trim();
      const dataLine = frame.match(/^data:\s*(.+)$/m)?.[1];
      if (!event || !dataLine) continue;
      try {
        yield { event, data: JSON.parse(dataLine) } as CompressEvent;
      } catch {
        // ignore malformed
      }
    }
  }
}
```

import { useEffect, useRef } from "react";
import { streamCompress } from "../lib/hermesRpc";
import { useChatStore } from "../store/chat";

export default function ContextOverflowBanner() {
  const lastPromptTokens = useChatStore(s => s.lastPromptTokens);
  const persistedSessionId = useChatStore(s => s.persistedSessionId);
  const compressionState = useChatStore(s => s.compressionState);
  const setCompressionState = useChatStore(s => s.setCompressionState);
  const isStreaming = useChatStore(s => s.isStreaming);
  // ... 现有 pct 计算 ...

  const triggeredRef = useRef(false);

  useEffect(() => {
    // debounce: 同一次"涨到 80%" 只触发一次. 切会话 / 压缩完 reset.
    if (pct < 80) {
      triggeredRef.current = false;
      return;
    }
    if (triggeredRef.current) return;
    if (compressionState === "compressing") return;
    if (isStreaming) return;
    if (!persistedSessionId) return;

    triggeredRef.current = true;
    setCompressionState("compressing", { text: "准备压缩..." });

    (async () => {
      try {
        for await (const ev of streamCompress({ ... })) {
          if (ev.event === "compress.progress") {
            setCompressionState("compressing", { text: ev.data.text });
          } else if (ev.event === "compress.completed") {
            setCompressionState("done", {
              headline: ev.data.headline,
              tokenLine: ev.data.token_line,
            });
          } else if (ev.event === "compress.failed") {
            setCompressionState("failed", { text: ev.data.error });
          }
        }
      } catch (e) {
        setCompressionState("failed", { text: String(e) });
      }
    })();
  }, [pct, persistedSessionId, compressionState, isStreaming]);

  // render: 80%+ 显 "鲶鱼检测到 86%, 自动压缩中..." (跟 modal 联动)
  if (pct < 80) return null;
  return (
    <div style={BANNER_BASE_YELLOW}>
      🗜️ 上下文 {pct.toFixed(0)}% — 鲶鱼自动压缩中 (modal 显进度)
    </div>
  );
}
